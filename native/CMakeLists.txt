cmake_minimum_required(VERSION 3.22)
project(psx_geometry LANGUAGES CXX)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE PATH "" FORCE)
set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
add_subdirectory(godot-cpp)
add_library(psx_geometry SHARED geometry.cpp spu_audio.cpp)
target_link_libraries(psx_geometry PRIVATE godot-cpp)
target_compile_features(psx_geometry PRIVATE cxx_std_17)
set_target_properties(psx_geometry PROPERTIES OUTPUT_NAME "psx_geometry_v29" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
add_executable(spu_dsp_test spu_dsp_test.cpp)
target_compile_features(spu_dsp_test PRIVATE cxx_std_17)
enable_testing()
add_test(NAME spu_dsp COMMAND spu_dsp_test)
